#   pip install --index-url https://download.pytorch.org/whl/cpu torch==2.14.0 torchvision==0.29.0 -c cpu_repro/requirements.txt

# fine because pip processes constraints together, but if you ever hit
# dependency-resolver errors, install numpy first, then tensorflow, then
# the rest.
